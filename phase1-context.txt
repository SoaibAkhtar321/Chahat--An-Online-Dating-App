-- Helper to safely compute active tier (expired/cancelled rows return 'free')
CREATE OR REPLACE FUNCTION public.get_active_subscription_tier(_user_id text)
RETURNS text
LANGUAGE plpgsql STABLE SECURITY DEFINER SET search_path = public, pg_temp AS $$
DECLARE
  _tier text;
BEGIN
  IF _user_id IS NULL THEN
    RETURN 'free';
  END IF;

  SELECT tier INTO _tier
  FROM public.user_subscriptions
  WHERE user_id = _user_id
    AND status = 'active'
    AND (expires_at IS NULL OR expires_at > now())
  ORDER BY
    CASE tier
      WHEN 'vip' THEN 3
      WHEN 'plus' THEN 2
      WHEN 'premium' THEN 1
      ELSE 0
    END DESC
  LIMIT 1;

  RETURN coalesce(_tier, 'free');
END $$;

REVOKE EXECUTE ON FUNCTION public.get_active_subscription_tier(text) FROM PUBLIC, anon;
GRANT EXECUTE ON FUNCTION public.get_active_subscription_tier(text) TO authenticated, service_role;

-- ---------- 2. Hardened record_swipe with server-side tier entitlements ----------
CREATE OR REPLACE FUNCTION public.record_swipe(_to_id text, _kind text)
RETURNS jsonb LANGUAGE plpgsql SECURITY DEFINER SET search_path = public, pg_temp AS $$
DECLARE
  _from_id text := auth.uid()::text;
  _today date := (timezone('Asia/Kolkata', now()))::date;
  _like_limit int := public.setting_int('daily_like_limit', 10);
  _prio_limit int := public.setting_int('daily_priority_like_limit', 3);
  _tier text;
  _is_paid_tier boolean;
  _likes int; _prio int;
  _me public.profiles; _them public.profiles;
  _existing text; _match_id uuid; _is_match boolean := false;
BEGIN
  IF _from_id IS NULL THEN RAISE EXCEPTION 'Authentication required'; END IF;
  IF _to_id IS NULL OR _from_id = _to_id THEN RAISE EXCEPTION 'Cannot swipe on own profile'; END IF;
  IF _kind IS NULL OR _kind NOT IN ('like','pass','priority') THEN RAISE EXCEPTION 'Invalid swipe kind'; END IF;

  -- Serialise swipes between the same pair so two simultaneous likes always produce a match.
  PERFORM pg_advisory_xact_lock(hashtextextended(least(_from_id,_to_id) || '|' || greatest(_from_id,_to_id), 0));

  SELECT * INTO _me FROM public.profiles WHERE id = _from_id;
  IF _me.id IS NULL OR _me.status <> 'active' THEN RAISE EXCEPTION 'Account not active'; END IF;
  SELECT * INTO _them FROM public.profiles WHERE id = _to_id;
  IF _them.id IS NULL OR _them.status <> 'active' OR public.is_blocked_pair(_from_id, _to_id)
     OR (_me.seeking <> '' AND _them.gender <> '' AND _them.gender <> _me.seeking) THEN
    RAISE EXCEPTION 'Profile not available';
  END IF;

  _tier := public.get_active_subscription_tier(_from_id);
  _is_paid_tier := _tier IN ('premium', 'plus', 'vip');

  SELECT kind INTO _existing FROM public.swipes WHERE from_id = _from_id AND to_id = _to_id;

  -- Quota applies only when this swipe newly spends a like / priority like.
  IF _kind IN ('like','priority') AND _existing IS DISTINCT FROM _kind THEN
    INSERT INTO public.user_daily_quotas (user_id, ist_date) VALUES (_from_id, _today)
      ON CONFLICT (user_id, ist_date) DO NOTHING;
    SELECT likes_used, super_likes_used INTO _likes, _prio FROM public.user_daily_quotas
      WHERE user_id = _from_id AND ist_date = _today FOR UPDATE;

    -- Free users strictly limited to 10 daily likes (T6 requirement)
    IF _kind = 'like' AND NOT _is_paid_tier AND _likes >= _like_limit THEN
      RETURN jsonb_build_object('success', false, 'error', 'daily_like_limit_reached',
        'likes_remaining', 0, 'reset_at_ist', _today + 1);
    END IF;

    IF _kind = 'priority' AND _prio >= _prio_limit THEN
      RETURN jsonb_build_object('success', false, 'error', 'daily_super_like_limit_reached',
        'super_likes_remaining', 0,
        'likes_remaining', (CASE WHEN _is_paid_tier THEN 999 ELSE greatest(0, _like_limit - _likes) END),
        'reset_at_ist', _today + 1);
    END IF;

    UPDATE public.user_daily_quotas
      SET likes_used = likes_used + (CASE WHEN _kind = 'like' THEN 1 ELSE 0 END),
          super_likes_used = super_likes_used + (CASE WHEN _kind = 'priority' THEN 1 ELSE 0 END),
          updated_at = now()
      WHERE user_id = _from_id AND ist_date = _today
      RETURNING likes_used, super_likes_used INTO _likes, _prio;
  ELSE
    SELECT likes_used, super_likes_used INTO _likes, _prio FROM public.user_daily_quotas
      WHERE user_id = _from_id AND ist_date = _today;
    _likes := coalesce(_likes, 0); _prio := coalesce(_prio, 0);
  END IF;

  INSERT INTO public.swipes (from_id, to_id, kind) VALUES (_from_id, _to_id, _kind)
    ON CONFLICT (from_id, to_id) DO UPDATE SET kind = EXCLUDED.kind, created_at = now();

  IF _kind <> 'pass' AND (
       EXISTS (SELECT 1 FROM public.swipes WHERE from_id = _to_id AND to_id = _from_id AND kind <> 'pass')
       OR (_them.is_demo AND _them.likes_you)) THEN
    INSERT INTO public.matches (user_a, user_b) VALUES (least(_from_id,_to_id), greatest(_from_id,_to_id))
      ON CONFLICT (user_a, user_b) DO NOTHING;
    SELECT id INTO _match_id FROM public.matches
      WHERE user_a = least(_from_id,_to_id) AND user_b = greatest(_from_id,_to_id);
    _is_match := true;
  END IF;

  RETURN jsonb_build_object('success', true, 'is_match', _is_match, 'match_id', _match_id,
    'likes_remaining', (CASE WHEN _is_paid_tier THEN 999 ELSE greatest(0, _like_limit - _likes) END),
    'super_likes_remaining', greatest(0, _prio_limit - _prio));
END $$;

  read_at timestamptz

  RETURN QUERY
  SELECT
    m.id,
    m.match_id,
    m.sender,
    m.body,
    m.image_path,
    m.created_at,

    CASE
      WHEN m.sender = _my_id AND _tier NOT IN ('plus', 'vip') THEN NULL
      ELSE m.read_at
    END AS read_at
  FROM public.messages m
  WHERE m.match_id = _match_id
  ORDER BY m.created_at ASC;

-- 3. Service-role-only RPC: record_verified_purchase -------------------------
-- Atomically records a verified store purchase and updates user_subscriptions.
-- If the token_hash was already redeemed, it raises an exception (replay protection).
CREATE OR REPLACE FUNCTION public.record_verified_purchase(
  _user_id text,
  _store text,
  _order_id text,
  _product_id text,
  _tier text,
  _token_hash text,
  _amount_inr numeric,
  _duration_months integer
)
RETURNS jsonb
LANGUAGE plpgsql SECURITY DEFINER SET search_path = public, pg_temp AS $$
DECLARE
  _uid uuid;
  _starts_at timestamptz := now();
  _expires_at timestamptz;
  _order_uuid uuid;
BEGIN
  IF _user_id IS NULL THEN RAISE EXCEPTION 'User ID required'; END IF;
  _uid := _user_id::uuid;

  IF _store IS NULL OR _store NOT IN ('google_play', 'stripe', 'manual_grant') THEN
    RAISE EXCEPTION 'Invalid store';
  END IF;

  IF _tier IS NULL OR _tier NOT IN ('premium', 'plus', 'vip') THEN
    RAISE EXCEPTION 'Invalid subscription tier';
  END IF;

  IF _token_hash IS NULL OR char_length(_token_hash) < 16 THEN
    RAISE EXCEPTION 'Invalid purchase token hash';
  END IF;

  IF _duration_months IS NULL OR _duration_months < 1 THEN
    RAISE EXCEPTION 'Duration must be at least 1 month';
  END IF;

  _expires_at := _starts_at + (_duration_months || ' months')::interval;

  -- Replay check: prevent duplicate token redemption
  IF EXISTS (SELECT 1 FROM public.subscription_orders WHERE store = _store AND token_hash = _token_hash) THEN
    RAISE EXCEPTION 'Purchase token has already been redeemed';
  END IF;

  -- Insert order record
  INSERT INTO public.subscription_orders (
    user_id, store, order_id, product_id, token_hash, tier, amount_inr, starts_at, expires_at
  ) VALUES (
    _uid, _store, _order_id, _product_id, _token_hash, _tier, coalesce(_amount_inr, 0.00), _starts_at, _expires_at
  ) RETURNING id INTO _order_uuid;

  -- Update or insert user_subscriptions record
  INSERT INTO public.user_subscriptions (
    user_id, tier, status, starts_at, expires_at, auto_renew, updated_at
  ) VALUES (
    _user_id, _tier, 'active', _starts_at, _expires_at, true, now()
  )
  ON CONFLICT (user_id) DO UPDATE SET
    tier = EXCLUDED.tier,
    status = 'active',
    starts_at = EXCLUDED.starts_at,
    expires_at = EXCLUDED.expires_at,
    auto_renew = EXCLUDED.auto_renew,
    updated_at = now();

  RETURN jsonb_build_object(
    'success', true,
    'order_id', _order_uuid,
    'tier', _tier,
    'expires_at', _expires_at
  );
END $$;

// Supabase Edge Function: verify-purchase
// Verifies store purchase tokens and records subscriptions server-side with replay protection.
// STATUS: Deploy with: supabase functions deploy verify-purchase

import { createClient } from "https://esm.sh/@supabase/supabase-js@2.49.1";

const CORS = {
  "Access-Control-Allow-Origin": "*",
  "Access-Control-Allow-Headers": "authorization, x-client-info, apikey, content-type",
};

const json = (status: number, body: Record<string, unknown>) =>
  new Response(JSON.stringify(body), {
    status,
    headers: { ...CORS, "Content-Type": "application/json" },
  });

async function sha256Hex(message: string): Promise<string> {
  const msgUint8 = new TextEncoder().encode(message);
  const hashBuffer = await crypto.subtle.digest("SHA-256", msgUint8);
  const hashArray = Array.from(new Uint8Array(hashBuffer));
  return hashArray.map((b) => b.toString(16).padStart(2, "0")).join("");
}

Deno.serve(async (req) => {
  if (req.method === "OPTIONS") return new Response("ok", { headers: CORS });
  if (req.method !== "POST") return json(405, { success: false, error: "Method not allowed" });

  const url = Deno.env.get("SUPABASE_URL");
  const anon = Deno.env.get("SUPABASE_ANON_KEY");
  const serviceKey = Deno.env.get("SUPABASE_SERVICE_ROLE_KEY");
  if (!url || !anon || !serviceKey) {
    return json(500, { success: false, error: "Server misconfigured" });
  }

  const authHeader = req.headers.get("Authorization") ?? "";
  if (!authHeader.startsWith("Bearer ")) {
    return json(401, { success: false, error: "Authentication required" });
  }

  // 1. Identify the caller strictly from their verified JWT
  const userClient = createClient(url, anon, {
    global: { headers: { Authorization: authHeader } },
  });
  const { data: userData, error: userErr } = await userClient.auth.getUser();
  if (userErr || !userData.user) {
    return json(401, { success: false, error: "Invalid or expired session" });
  }
  const uid = userData.user.id;

  // 2. Parse and validate payload
  let body: any = {};
  try {
    body = await req.json();
  } catch {
    return json(400, { success: false, error: "Invalid JSON body" });
  }

  const { store, orderId, productId, purchaseToken, tier, amountInr, durationMonths } = body;

  const validStores = ["google_play", "stripe", "manual_grant"];
  if (!store || !validStores.includes(store)) {
    return json(400, { success: false, error: "Invalid or unsupported store" });
  }

  const validTiers = ["premium", "plus", "vip"];
  if (!tier || !validTiers.includes(tier)) {
    return json(400, { success: false, error: "Invalid subscription tier" });
  }

  if (!productId || typeof productId !== "string") {
    return json(400, { success: false, error: "Product ID is required" });
  }

  if (!purchaseToken || typeof purchaseToken !== "string" || purchaseToken.trim().length < 16) {
    return json(400, { success: false, error: "Valid purchase token is required" });
  }

  const months = Number(durationMonths);
  if (!months || isNaN(months) || months < 1) {
    return json(400, { success: false, error: "Duration must be at least 1 month" });
  }

  // 3. Compute token hash for cryptographic replay protection
  const tokenHash = await sha256Hex(purchaseToken.trim());

  // 4. Record purchase atomically using privileged service_role client
  const admin = createClient(url, serviceKey, {
    auth: { persistSession: false, autoRefreshToken: false },
  });

  const { data: rpcRes, error: rpcErr } = await admin.rpc("record_verified_purchase", {
    _user_id: uid,
    _store: store,
    _order_id: orderId || null,
    _product_id: productId,
    _tier: tier,
    _token_hash: tokenHash,
    _amount_inr: Number(amountInr) || 0,
    _duration_months: months,
  });

  if (rpcErr) {
    const isReplay = rpcErr.message?.includes("already been redeemed");
    return json(isReplay ? 409 : 400, {
      success: false,
      error: rpcErr.message || "Failed to record verified purchase",
    });
  }

  return json(200, {
    success: true,
    data: rpcRes,
  });
});

#!/usr/bin/env bash
# One-command database security/regression suite on a SCRATCH PostgreSQL 16 (never touches Supabase).
# Requires: psql + a local disposable Postgres instance with the sentinel comment.
# Exit code is non-zero if anything fails. SQL NOTICEs are not failures.
set -u
cd "$(dirname "$0")"

# 1. Environment Sanitization: prevent inherited libpq variables from redirecting connections
unset PGDATABASE PGSERVICE PGSERVICEFILE DATABASE_URL PGOPTIONS PGSSLMODE

# 2. Connection Parameters
export PGHOST="${PGHOST:-localhost}" PGPORT="${PGPORT:-5433}" PGUSER="${PGUSER:-postgres}"

case "$PGHOST" in
  localhost|127.0.0.1|::1|/*) ;;
  *) [ "${ALLOW_NONLOCAL_DB:-}" = "1" ] || { echo "REFUSING to run against non-local PGHOST=$PGHOST (set ALLOW_NONLOCAL_DB=1 only for a disposable server)"; exit 2; } ;;
esac

# Guard against accidental default port targeting
if [ "$PGPORT" = "5432" ] && [ "${ALLOW_DANGEROUS_DEFAULT_PORT:-0}" != "1" ]; then
  echo "FAIL-CLOSED: Refusing to run tests against default PostgreSQL port 5432." >&2
  echo "Tests must target a dedicated container port (e.g. PGPORT=5433). Set ALLOW_DANGEROUS_DEFAULT_PORT=1 to override." >&2
  exit 2
fi

# 3. Dedicated Administrative Wrapper bound strictly to 'postgres' database
psql_admin() {
  psql -X -q -v ON_ERROR_STOP=1 -h "$PGHOST" -p "$PGPORT" -U "$PGUSER" -d postgres "$@"
}

# 4. Target Connection Handshake
TARGET_INFO=$(psql_admin -At -c "SELECT current_user, current_database();" 2>/dev/null) || {
  echo "FAIL-CLOSED: Could not connect to PostgreSQL target on ${PGHOST}:${PGPORT} as user ${PGUSER}." >&2
  exit 2
}

  echo "FAIL-CLOSED: Unexpected connection target: '$TARGET_INFO' (expected ${PGUSER}|postgres)." >&2
  exit 2
fi

# 5. Supplementary Sentinel Check on 'postgres' database
SENTINEL=$(psql_admin -At -c "SELECT coalesce(shobj_description(oid, 'pg_database'), '') FROM pg_database WHERE datname = 'postgres';" 2>/dev/null)
if [ "$SENTINEL" != "milo_disposable_test_cluster" ]; then
  echo "FAIL-CLOSED: Target PostgreSQL cluster lacks 'milo_disposable_test_cluster' sentinel on database 'postgres'." >&2
  echo "Refusing destructive test run against unverified cluster." >&2
  exit 2
fi

# 6. Strict Initial Allowlist Check: only standard system databases allowed before test creation
UNEXPECTED_DBS=$(psql_admin -At -c "
  SELECT datname FROM pg_database 
  WHERE datname NOT IN ('postgres', 'template0', 'template1');
" 2>/dev/null)

if [ -n "$UNEXPECTED_DBS" ]; then
  echo "FAIL-CLOSED: Target cluster contains foreign or uncleaned database(s):" >&2
  echo "$UNEXPECTED_DBS" >&2
  echo "A clean disposable cluster must contain ONLY standard system databases ('postgres', 'template0', 'template1')." >&2
  exit 2
fi

# 7. Collision-Resistant Run Identifier & Database Names
RAND_HEX=$(od -An -N4 -tx4 /dev/urandom 2>/dev/null | tr -d ' ' || echo "$RANDOM$RANDOM")
RUN_ID="m_${RAND_HEX}_${$}_$(date +%s)"
FUNC_DB="${RUN_ID}_f"
CONC_DB="${RUN_ID}_c"

if [ "${#FUNC_DB}" -gt 63 ] || [ "${#CONC_DB}" -gt 63 ]; then
  echo "FATAL: Database identifier exceeds PostgreSQL 63-byte limit." >&2
  exit 2
fi
if ! [[ "$FUNC_DB" =~ ^[a-z_][a-z0-9_]*$ ]] || ! [[ "$CONC_DB" =~ ^[a-z_][a-z0-9_]*$ ]]; then
  echo "FATAL: Database identifier contains invalid characters." >&2
  exit 2
fi

# Pre-existing Database Collision Check
EXISTS_CHECK=$(psql_admin -At -c "SELECT count(*) FROM pg_database WHERE datname IN ('$FUNC_DB', '$CONC_DB');" 2>/dev/null)
if [ "$EXISTS_CHECK" != "0" ]; then
  echo "FAIL-CLOSED: Database collision detected for '$FUNC_DB' or '$CONC_DB'. Aborting without overwrite." >&2
  exit 2
fi

# 8. Overflow-Safe Concurrency Lock Key Generation via PostgreSQL Engine
LOCK_KEY=$(psql_admin -At -c "SELECT (abs(hashtext('$RUN_ID')::bigint) % 2000000000) + 1000;" 2>/dev/null)
if ! [[ "$LOCK_KEY" =~ ^[1-9][0-9]{3,9}$ ]] || [ "$LOCK_KEY" -gt 2000000999 ]; then
  echo "FATAL: Generated invalid lock key: '$LOCK_KEY'" >&2
  exit 2
fi

# 9. Test Execution Configuration
MIG="${MIGRATIONS_DIR:-../migrations}"

psql_test() {
  local target_db="$1"; shift
  psql -X -q -v ON_ERROR_STOP=1 -h "$PGHOST" -p "$PGPORT" -U "$PGUSER" -d "$target_db" "$@"
}

setup_db() {
  local db="$1"
  psql_admin -c "CREATE DATABASE \"$db\"" 2>&1 | grep -v NOTICE
  psql_test "$db" -f 00_supabase_standins.sql || { echo "FAIL: stand-ins"; return 1; }
  local files=("$MIG"/*.sql); [ -f ../seed.sql ] && files+=(../seed.sql)
  [ -e "${files[0]}" ] || { echo "FAIL: no migrations found in $MIG"; return 1; }
  for f in "${files[@]}"; do
    local err_file="/tmp/${RUN_ID}_apply_${f##*/}.err"
    psql_test "$db" -f "$f" >/dev/null 2>"$err_file" || {
      echo "FAIL: applying migration ${f##*/}" >&2
      cat "$err_file" >&2
      return 1
    }
  done
  echo "applied migrations + seed to $db"
}

setup_db "$FUNC_DB" || exit 1
for f in 10_functional_rls.sql 20_storage_deletion.sql 25_photo_reviews.sql 27_photo_visibility.sql 28_security_foundations.sql 35_uniform_quota.sql 90_assert.sql; do

  if [ $rc -ne 0 ]; then
    echo "FAIL: $f (exit code $rc)" >&2
    echo "=== Failure Output ($out_file) ===" >&2
    cat "$out_file" >&2
    echo "==================================" >&2
    echo "ABORTING: Preserving databases '$FUNC_DB' and '$CONC_DB'. Concurrency suite skipped." >&2
    exit $rc
  fi
  [ "$f" = 90_assert.sql ] && cat "$out_file"

echo "== concurrency =="
setup_db "$CONC_DB" || exit 1
psql_test "$CONC_DB" -f 30_concurrency_setup.sql || { echo "FAIL: concurrency setup"; exit 1; }
bash 40_concurrency.sh "$CONC_DB" "$LOCK_KEY"

  echo "FAIL: Concurrency tests failed (exit code $conc_rc)" >&2
  echo "ABORTING: Preserving databases '$FUNC_DB' and '$CONC_DB'." >&2
  exit $conc_rc
fi

echo "All tests passed cleanly. Cleaning up run-specific databases..."
psql_admin -c "DROP DATABASE \"$FUNC_DB\";" -c "DROP DATABASE \"$CONC_DB\";" >/dev/null 2>&1
cleanup_rc=$?
if [ $cleanup_rc -ne 0 ]; then
  echo "WARNING: All tests passed, but cleanup failed to drop databases." >&2
  echo "Databases retained on cluster: '$FUNC_DB', '$CONC_DB'" >&2
  exit 3
fi

echo "RESULT: ALL DATABASE TESTS PASSED"
exit 0

-- Fails the run (non-zero exit) if any check failed or if the suite looks empty.

DECLARE failed int; total int;
BEGIN
  SELECT count(*) FILTER (WHERE NOT ok), count(*) INTO failed, total FROM public.t_results;
  IF total < 50 THEN RAISE EXCEPTION 'suite looks incomplete: only % checks recorded', total; END IF;
  IF failed > 0 THEN RAISE EXCEPTION '% of % checks FAILED', failed, total; END IF;

===== supabase\tests\README.md =====
# Database security suite

Entry point: `bash supabase/tests/run.sh` (needs `psql` and a **disposable** local PostgreSQL 16; it drops and recreates `milo_test_func` and `milo_test_conc`). It refuses a non-local `PGHOST` unless `ALLOW_NONLOCAL_DB=1`.
Env: `PGHOST` (default `localhost`; may be a socket dir), `PGPORT`, `PGUSER`, `PGPASSWORD`, `MIGRATIONS_DIR` (default `../migrations`, applied in lexical order; `../seed.sql` is applied only if it exists).
Order: stand-ins, migrations, `10_functional_rls.sql`, `20_storage_deletion.sql`, `90_assert.sql` (needs at least 50 recorded checks), then a fresh DB with `30_concurrency_setup.sql` and `40_concurrency.sh`.

## What the stand-ins (`00_supabase_standins.sql`) reproduce
Roles `anon`/`authenticated`/`service_role`, `auth.users`, `auth.uid()`/`auth.role()` driven by `request.jwt.claim.*` settings, default table/function grants to those roles, minimal `storage.buckets`/`storage.objects`/`storage.foldername`, and a `supabase_realtime` publication.

## What they do NOT reproduce
Real Supabase Auth (JWT verification, `auth.users` columns/triggers), the real Storage service and its policies/buckets, PostgREST behaviour, Supabase's real default privileges, extensions and roles, and the **actual remote schema**. A pass here is not evidence that the Supabase project is secure; a fail shows the committed SQL does not meet the suite's expectations.

## Known state
The migration history is unified and timestamp-ordered in `supabase/migrations/`:
1. `20260101000000_core_schema.sql`
2. `20260101000100_storage.sql`
3. `20260101000200_admin_moderation.sql`
4. `20260101000300_mobile_security_hardening.sql`
5. `20260101000400_security_followups.sql`
6. `20260101000500_privacy_and_rate_limits.sql`
7. `20260101000600_trigger_function_hardening.sql`

Legacy Drizzle migration files in `archive/legacy-drizzle/drizzle/migrations/` are superseded, archived, and kept for historical reference only. CI runs directly against `supabase/migrations/` and `supabase/seed.sql`.
